cmake_minimum_required(VERSION 3.25)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Hardware IOUSBHost bench minimum" FORCE)
project(VirusHardwareBench LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_AUV2_BUILDS OFF CACHE BOOL "" FORCE)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/vst3sdk")
set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
add_subdirectory("${SDK_ROOT}" sdk-build)
add_executable(hardware-bench main.mm ../usb-descriptors/src/usb_descriptors.cpp
 "${SDK_ROOT}/public.sdk/source/vst/hosting/module_mac.mm"
 "${SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp")
set_source_files_properties(main.mm "${SDK_ROOT}/public.sdk/source/vst/hosting/module_mac.mm" PROPERTIES COMPILE_FLAGS "-fobjc-arc")
target_include_directories(hardware-bench PRIVATE "${SDK_ROOT}" ../device-bridge/include ../transport-core/include ../usb-descriptors/include ../midi-core/include)
target_link_libraries(hardware-bench PRIVATE sdk_hosting "-framework Cocoa" "-framework CoreAudio" "-framework CoreMIDI" "-framework IOKit" "-framework IOUSBHost")
target_compile_definitions(hardware-bench PRIVATE "VTR_PLUGIN_PATH=\"${CMAKE_CURRENT_SOURCE_DIR}/../vst3/build/VST3/Release/Virus TI Rework Development.vst3\"")
