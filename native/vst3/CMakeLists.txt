cmake_minimum_required(VERSION 3.25)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Native Apple silicon")
set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "macOS minimum")
project(VirusTIRework VERSION 0.1.0 LANGUAGES C CXX OBJCXX)
option(VTR_SANITIZE "Build offline tests with sanitizers" OFF)
if(VTR_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "Never install globally" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "Run explicitly" FORCE)
set(SMTG_ENABLE_AUV2_BUILDS OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_IOS_TARGETS OFF CACHE BOOL "" FORCE)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/vst3sdk")
set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
add_subdirectory("${SDK_ROOT}" sdk-build)
smtg_add_vst3plugin(VirusTIRework PACKAGE_NAME "Virus TI Rework Development" SOURCES_LIST plugin.cpp plugin.h catalog.h ../editor/editor.mm ../editor/editor.h)
target_include_directories(VirusTIRework PRIVATE "${SDK_ROOT}" "../device-bridge/include" "../transport-core/include" "../midi-core/include" "../editor" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(VirusTIRework PRIVATE sdk "-framework Cocoa")
smtg_target_set_bundle(VirusTIRework BUNDLE_IDENTIFIER "org.virus-ti-rework.development.vst3" COMPANY_NAME "Virus TI Rework Project")
add_executable(vst3_processor_tests tests.cpp plugin.cpp ../editor/editor.mm "${SDK_ROOT}/public.sdk/source/common/memorystream.cpp" "${SDK_ROOT}/public.sdk/source/vst/hosting/eventlist.cpp" "${SDK_ROOT}/public.sdk/source/vst/hosting/parameterchanges.cpp")
target_include_directories(vst3_processor_tests PRIVATE "${SDK_ROOT}" "../device-bridge/include" "../transport-core/include" "../midi-core/include" "../editor" "${CMAKE_CURRENT_SOURCE_DIR}")
target_sources(vst3_processor_tests PRIVATE "../transport-core/src/transport_core.cpp")
target_compile_definitions(vst3_processor_tests PRIVATE VIRUS_REWORK_TEST=1)
target_link_libraries(vst3_processor_tests PRIVATE sdk sdk_hosting "-framework CoreFoundation" "-framework Cocoa")
enable_testing()
add_test(NAME vst3_processor_tests COMMAND vst3_processor_tests)

set_target_properties(VirusTIRework PROPERTIES
 MACOSX_BUNDLE_GUI_IDENTIFIER "org.virus-ti-rework.development.vst3"
 MACOSX_BUNDLE_BUNDLE_NAME "Virus TI Rework (Development)"
 MACOSX_BUNDLE_BUNDLE_VERSION "0.1.0"
 MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1.0")

set_source_files_properties(../editor/editor.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
set(ORIGINAL_RESOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../../analysis/package-expanded/Core_components.pkg/Payload/Library/Application Support/Access Music/Virus TI/Common/Virus Control.bundle/Contents/Resources")
file(GLOB ORIGINAL_PNGS "${ORIGINAL_RESOURCES}/*.png")
set(EDITOR_CATALOG "${CMAKE_CURRENT_SOURCE_DIR}/../../analysis/editor-ui-catalog.json")
set_source_files_properties(${ORIGINAL_PNGS} ${EDITOR_CATALOG} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/OriginalEditor")
target_sources(VirusTIRework PRIVATE ${ORIGINAL_PNGS} ${EDITOR_CATALOG})
add_executable(vst3_editor_tests ../editor/editor_tests.mm plugin.cpp)
set_source_files_properties(../editor/editor_tests.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
target_include_directories(vst3_editor_tests PRIVATE "${SDK_ROOT}" "../device-bridge/include" "../transport-core/include" "../midi-core/include" "../editor" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(vst3_editor_tests PRIVATE VIRUS_REWORK_TEST=1)
target_link_libraries(vst3_editor_tests PRIVATE sdk "-framework Cocoa")
add_test(NAME vst3_editor_tests COMMAND vst3_editor_tests "$<TARGET_BUNDLE_DIR:VirusTIRework>/Contents/Resources/OriginalEditor" "${CMAKE_CURRENT_BINARY_DIR}/editor-renders")

set(EDITOR_MODULES ../editor/graphs.mm ../editor/browser.mm ../editor/browser_bank.cpp)
set_source_files_properties(../editor/graphs.mm ../editor/browser.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
foreach(editor_target VirusTIRework vst3_processor_tests vst3_editor_tests)
 target_sources(${editor_target} PRIVATE ${EDITOR_MODULES})
endforeach()

set(PATCH_BANK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../analysis/package-expanded/Additional_items.pkg/Payload/Library/Application Support/Access Music/Virus TI/Patches")
file(GLOB_RECURSE PATCH_BANKS "${PATCH_BANK_ROOT}/*.mid")
foreach(bank ${PATCH_BANKS})
 file(RELATIVE_PATH relative_bank "${PATCH_BANK_ROOT}" "${bank}")
 get_filename_component(bank_subdirectory "${relative_bank}" DIRECTORY)
 set_source_files_properties("${bank}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/PatchBanks/${bank_subdirectory}")
endforeach()
target_sources(VirusTIRework PRIVATE ${PATCH_BANKS})

set_source_files_properties(hardware_service.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
foreach(target VirusTIRework vst3_processor_tests vst3_editor_tests)
 target_sources(${target} PRIVATE hardware_service.mm ../usb-descriptors/src/usb_descriptors.cpp)
 target_include_directories(${target} PRIVATE ../usb-descriptors/include)
 target_link_libraries(${target} PRIVATE "-framework CoreAudio" "-framework CoreMIDI" "-framework IOKit" "-framework IOUSBHost")
endforeach()
set(STOCK_PROBE "${CMAKE_CURRENT_SOURCE_DIR}/../../analysis/hardware/fixtures/stock-wavetable-probe.syx")
set_source_files_properties(${STOCK_PROBE} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/Hardware")
target_sources(VirusTIRework PRIVATE ${STOCK_PROBE})

add_executable(capture_fifo_tests capture_fifo_tests.cpp)
target_compile_options(capture_fifo_tests PRIVATE -UNDEBUG)
add_test(NAME capture_fifo_tests COMMAND capture_fifo_tests)

add_executable(local_control_tests local_control_tests.cpp)
target_compile_options(local_control_tests PRIVATE -UNDEBUG)
add_test(NAME local_control_tests COMMAND local_control_tests)

foreach(target VirusTIRework vst3_editor_tests)
 target_sources(${target} PRIVATE "${SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
endforeach()

# The optional emulator runtime is built independently from the pinned OsTIrus source.
option(VTR_BUNDLE_EMULATOR "Bundle separately built emulator runtime (separate licensing)" OFF)
set(EMULATOR_RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../osti-render/build/libosti-backend.dylib")
if(VTR_BUNDLE_EMULATOR AND EXISTS "${EMULATOR_RUNTIME}")
 set_source_files_properties("${EMULATOR_RUNTIME}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/Emulator")
 target_sources(VirusTIRework PRIVATE "${EMULATOR_RUNTIME}")
endif()
add_executable(vst3_api_host api_host.cpp plugin.cpp ../editor/editor.mm hardware_service.mm ../usb-descriptors/src/usb_descriptors.cpp ${EDITOR_MODULES} "${SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
target_compile_definitions(vst3_api_host PRIVATE VIRUS_REWORK_TEST=1)
target_include_directories(vst3_api_host PRIVATE "${SDK_ROOT}" ../device-bridge/include ../transport-core/include ../midi-core/include ../editor ../usb-descriptors/include "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(vst3_api_host PRIVATE sdk "-framework Cocoa" "-framework CoreAudio" "-framework CoreMIDI" "-framework IOKit" "-framework IOUSBHost")
foreach(target VirusTIRework vst3_processor_tests vst3_editor_tests vst3_api_host)
 target_compile_definitions(${target} PRIVATE VTR_BUNDLE_RESOURCES="$<TARGET_BUNDLE_DIR:VirusTIRework>/Contents/Resources")
endforeach()

add_executable(meter_tests meter_tests.cpp)
target_compile_options(meter_tests PRIVATE -UNDEBUG)
add_test(NAME meter_tests COMMAND meter_tests)

add_executable(virus-ti-restoration-helper restoration_helper.mm ../editor/browser_bank.cpp ../usb-descriptors/src/usb_descriptors.cpp)
set_source_files_properties(restoration_helper.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
target_include_directories(virus-ti-restoration-helper PRIVATE ../editor ../midi-core/include ../usb-descriptors/include)
target_link_libraries(virus-ti-restoration-helper PRIVATE "-framework Foundation" "-framework CoreMIDI" "-framework IOKit" "-framework IOUSBHost")
add_dependencies(VirusTIRework virus-ti-restoration-helper)
add_custom_command(TARGET VirusTIRework POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:VirusTIRework>/Contents/Resources/Helpers"
 COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:virus-ti-restoration-helper>" "$<TARGET_BUNDLE_DIR:VirusTIRework>/Contents/Resources/Helpers/virus-ti-restoration-helper"
 COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:VirusTIRework>/Contents/Resources/Helpers/virus-ti-restoration-helper"
 COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:VirusTIRework>")

add_executable(restoration_tests restoration_tests.cpp ../editor/browser_bank.cpp)
target_include_directories(restoration_tests PRIVATE ../editor)
target_compile_definitions(restoration_tests PRIVATE STOCK_PROBE="${STOCK_PROBE}" RESTORE_HELPER="$<TARGET_FILE:virus-ti-restoration-helper>")
add_dependencies(restoration_tests virus-ti-restoration-helper)
target_compile_options(restoration_tests PRIVATE -UNDEBUG)
add_test(NAME restoration_tests COMMAND restoration_tests)
